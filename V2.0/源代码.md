// ================== 主逻辑 ==================
function update(dt){
  // 等级与下落速度随分数提升
  level=1+Math.floor(sc/60);
  var fall=(2+ (level-1)*0.35)*(dt*60);
  // 移动（键盘或按钮任一按住即生效）
  var ax=0;
  if((keyL||btnL)&&!dragging)ax-=v;
  if((keyR||btnR)&&!dragging)ax+=v;
  beanMove=ax?1:0;
  vx+= (ax-vx)*Math.min(dt*12,1);
  x+=vx*(dt*60);
  if(x<22){x=22;vx=0;} if(x>W-22){x=W-22;vx=0;}
  bob+=dt*4;
  // 计时器
  if(invT>0)invT-=dt;
  if(magT>0)magT-=dt;
  if(starT>0)starT-=dt;
  if(comboT>0){comboT-=dt;if(comboT<=0)combo=0;}
  if(shake>0)shake-=dt;
  // 生成（修复：用计时器，不再依赖 y==80 精确相等）
  spawnT-=dt;
  var interval=Math.max(0.45,1.1-level*0.06);
  if(spawnT<=0&&it.length<18){addItem();spawnT=interval;}
  // 掉落物
  for(var i=it.length-1;i>=0;i--){
    var o=it[i];
    o.y+=fall;o.rot+=dt*2;
    // 磁铁：吸附范围内加分物
    if(magT>0&&TYPES[o.t].good>=1){
      var dx=x-o.x,dy=y-o.y,d=Math.hypot(dx,dy);
      if(d<120){o.x+=dx/d*4.5*(dt*60)*0.35;o.y+=dy/d*4.5*(dt*60)*0.35;}
    }
    if(Math.hypot(o.x-x,o.y-y)<30){
      switch(o.t){
        case"meat1":combo++;comboT=3;addScore(1,o.x,o.y);SND.eat();burst(o.x,o.y,"#FFD300",8);break;
        case"meat5":combo++;comboT=3;addScore(5,o.x,o.y);SND.five();burst(o.x,o.y,"#FF2B2B",12,2.8);break;
        case"gold": combo++;comboT=3;addScore(1,o.x,o.y);coin++;save();SND.coin();burst(o.x,o.y,"#ffd700",14,3);break;
        case"poo":  if(invT<=0){sc=Math.max(0,sc-1);save();floatText(o.x,o.y,"-1 分","#8a6d3b",14);} takeHit(o.x,o.y);break;
        case"bomb": takeHit(o.x,o.y);break;
        case"light":invT=Math.max(invT,0)+6;SND.power();floatText(o.x,o.y,"无敌 6 秒！","#30A8FF",15);burst(o.x,o.y,"#30A8FF",14);break;   // 修复：统一计时，可叠加
        case"magnet":magT=8;SND.power();floatText(o.x,o.y,"磁铁 8 秒！","#ff2d95",15);burst(o.x,o.y,"#ff2d95",14);break;
        case"star": starT=10;SND.power();floatText(o.x,o.y,"双倍 10 秒！","#ffd100",15);burst(o.x,o.y,"#ffd100",14);break;
        case"shieldIt": shield=true;SND.power();floatText(o.x,o.y,"获得护盾！","#9be7ff",15);burst(o.x,o.y,"#9be7ff",14);break;
      }
      it.splice(i,1);
      continue;
    }
    if(o.y>H+sz)it.splice(i,1);   // 落出屏幕移除，防止数组堆积
  }
  // 粒子
  for(var j=particles.length-1;j>=0;j--){
    var p=particles[j];
    p.x+=p.vx*(dt*60);p.y+=p.vy*(dt*60);p.vy+=0.06*(dt*60);p.life-=dt*60;
    if(p.life<=0)particles.splice(j,1);
  }
  for(var k2=floats.length-1;k2>=0;k2--){
    var f=floats[k2];f.y-=0.9*(dt*60);f.life-=dt*60;
    if(f.life<=0)floats.splice(k2,1);
  }
}

// 立体球体：径向渐变+高光+边缘反光
function sphere3d(px,py,r,colTop,colMid,colBottom,hl){
  var g=ctx.createRadialGradient(px-r*0.35,py-r*0.4,r*0.15,px,py,r);
  g.addColorStop(0,colTop);
  g.addColorStop(0.55,colMid);
  g.addColorStop(1,colBottom);
  ctx.fillStyle=g;
  ctx.beginPath();ctx.arc(px,py,r,0,6.283);ctx.fill();
  // 边缘环境反光
  ctx.strokeStyle="rgba(255,255,255,0.28)";ctx.lineWidth=r*0.09;
  ctx.beginPath();ctx.arc(px,py,r*0.92,Math.PI*0.7,Math.PI*1.45);ctx.stroke();
  // 镜面高光
  ctx.fillStyle="rgba(255,255,255,"+(hl||0.75)+")";
  ctx.beginPath();ctx.ellipse(px-r*0.34,py-r*0.42,r*0.26,r*0.16,-0.6,0,6.283);ctx.fill();
}

}

function drawStarShape(px,py,ro,ri,col){
  ctx.beginPath();
  for(var i=0;i<10;i++){
    var an=-Math.PI/2+i*Math.PI/5,r2=i%2?ri:ro;
    var px2=px+Math.cos(an)*r2,py2=py+Math.sin(an)*r2;
    i?ctx.lineTo(px2,py2):ctx.moveTo(px2,py2);
  }
  ctx.closePath();ctx.fillStyle=col;ctx.fill();
}
function drawShieldShape(px,py,h,col){
  ctx.beginPath();
  ctx.moveTo(px,py-h);
  ctx.lineTo(px+h*0.85,py-h*0.5);
  ctx.lineTo(px+h*0.7,py+h*0.55);
  ctx.lineTo(px,py+h);
  ctx.lineTo(px-h*0.7,py+h*0.55);
  ctx.lineTo(px-h*0.85,py-h*0.5);
  ctx.closePath();ctx.fillStyle=col;ctx.fill();
}

// ================== 渲染主循环 ==================
function draw(){
  ctx.clearRect(0,0,W,H);
  drawBg();
  if(state===ST.MENU)return;
  // 粒子（画在角色后面）
  particles.forEach(function(p){
    ctx.globalAlpha=Math.min(1,p.life/20);
    ctx.fillStyle=p.c;
    if(p.star){drawStarShape(p.x,p.y,p.s+1,p.s*0.45,p.c);}
    else{ctx.beginPath();ctx.arc(p.x,p.y,p.s,0,6.283);ctx.fill();}
  });
  ctx.globalAlpha=1;
  it.forEach(drawItem);
  drawBean();
  // 浮动文字
  floats.forEach(function(f){
    ctx.globalAlpha=Math.min(1,f.life/24);
    boldText(f.txt,f.x,f.y,f.size,f.c);
  });
  ctx.globalAlpha=1;
  drawHud();
  if(state===ST.PAUSE){
    ctx.fillStyle="rgba(90,50,10,0.4)";ctx.fillRect(0,0,W,H);
    boldText("已暂停 · 按 P 或点「继续」",W/2,H/2,20,"#fff");
  }
  if(state===ST.OVER)drawOver();
}

function loop(ts){
  requestAnimationFrame(loop);
  // 真实时间差驱动（修复：掉帧时速度不一致/生成逻辑漂移）
  var dt=lastT?Math.min((ts-lastT)/1000,0.05):0.016;
  lastT=ts;
  if(state===ST.PLAY)update(dt);
  if(shake>0){
    ctx.save();
    ctx.translate((Math.random()-0.5)*8*shake*2.5,(Math.random()-0.5)*8*shake*2.5);
  }
  draw();
  if(shake>0)ctx.restore();
}
requestAnimationFrame(loop);
